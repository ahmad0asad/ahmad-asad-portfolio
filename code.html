mask_s2_scl.js

Sentinel-2 Level-2A Cloud & Shadow Masking via SCL Band

Filters out cloud shadows, medium/high probability clouds, cirrus, and snow/ice using the Scene Classification Layer (SCL) from Sentinel-2 Surface Reflectance (SR) Harmonized products.

JavaScript (Google Earth Engine API) 
/**
 * Sentinel-2 Level-2A Cloud, Shadow & Cirrus Masking Function
 * Author: Ahmad Asad (PhD Scholar · University of Bari Aldo Moro)
 * Collection: COPERNICUS/S2_SR_HARMONIZED
 */
function maskS2sr(image) {
  // Select the Scene Classification Layer (SCL) band
  var scl = image.select('SCL');
  
  // SCL pixel classes to mask:
  // 3 = Cloud shadows, 8 = Cloud medium probability,
  // 9 = Cloud high probability, 10 = Thin cirrus, 11 = Snow / Ice
  var mask = scl.neq(3)
    .and(scl.neq(8))
    .and(scl.neq(9))
    .and(scl.neq(10))
    .and(scl.neq(11));
  
  // Also mask saturated or defective pixels (class 1)
  mask = mask.and(scl.neq(1));
  
  // Scale optical reflectances by 10,000 to standard 0.0 - 1.0 range
  var opticalBands = image.select('B.*').multiply(0.0001);
  
  return image.addBands(opticalBands, null, true)
    .updateMask(mask)
    .copyProperties(image, ['system:time_start', 'system:index']);
}

// Example Execution Pipeline:
var geometry = ee.Geometry.Point([16.8719, 41.1171]); // Bari, Italy
var s2Collection = ee.ImageCollection('COPERNICUS/S2_SR_HARMONIZED')
  .filterBounds(geometry)
  .filterDate('2025-05-01', '2025-09-30')
  .filter(ee.Filter.lt('CLOUDY_PIXEL_PERCENTAGE', 25))
  .map(maskS2sr);

var medianComposite = s2Collection.median();
Map.centerObject(geometry, 12);
Map.addLayer(medianComposite, {bands: ['B4', 'B3', 'B2'], min: 0, max: 0.3}, 'S2 Cloud-Free True Color');

compute_sui_stack.js

Sealed Urban Index (SUI) & Multi-Index Stack Computation

Calculates the Sealed Urban Index (SUI = Green - SWIR1) alongside NDVI, NDBI, and MNDWI on Sentinel-2, generating a multi-band feature stack calibrated for Random Forest classification.

JavaScript (Google Earth Engine API) 
/**
 * Sealed Urban Index (SUI) Multi-Spectral Stack Pipeline
 * Reference: Asad et al. (IJRS 2026) DOI: 10.1080/01431161.2026.2715141
 * Formula: SUI = Green (B3) - SWIR1 (B11)
 */
function addSpectralIndices(image) {
  // SUI: High values over sealed/impervious surfaces, negative over bare soil
  var green = image.select('B3');
  var red = image.select('B4');
  var nir = image.select('B8');
  var swir1 = image.select('B11');
  
  var sui = green.subtract(swir1).rename('SUI');
  var ndvi = image.normalizedDifference(['B8', 'B4']).rename('NDVI');
  var ndbi = image.normalizedDifference(['B11', 'B8']).rename('NDBI');
  var mndwi = image.normalizedDifference(['B3', 'B11']).rename('MNDWI');
  
  return image.addBands([sui, ndvi, ndbi, mndwi]);
}

// Apply across a cloud-masked summer collection
var aoi = ee.Geometry.Rectangle([16.70, 41.05, 17.05, 41.25]); // Bari Metro
var summerMosaic = ee.ImageCollection('COPERNICUS/S2_SR_HARMONIZED')
  .filterBounds(aoi)
  .filterDate('2025-06-01', '2025-08-31')
  .filter(ee.Filter.lt('CLOUDY_PIXEL_PERCENTAGE', 20))
  .median()
  .clip(aoi);

var stackedImage = addSpectralIndices(summerMosaic);

// SUI Visualization: Impervious surfaces emerge as high/bright values
Map.addLayer(stackedImage.select('SUI'), {min: -0.3, max: 0.15, palette: ['#2563eb', '#f8fafc', '#dc2626']}, 'Sealed Urban Index (SUI)');
Map.addLayer(stackedImage.select('NDVI'), {min: 0.0, max: 0.7, palette: ['#78350f', '#fde047', '#15803d']}, 'NDVI');

windowed_index_calc.py

Windowed GeoTIFF Index Calculation with NoData Handling

Computes spectral indices on multi-gigabyte satellite rasters using chunked block windows. Guarantees zero out-of-memory (OOM) crashes and preserves projection, spatial transforms, and compression.

Python (rasterio + numpy) 
"""
Windowed Spectral Index Processor with Memory Safety
Author: Ahmad Asad | Remote Sensing & GeoAI
Requirements: pip install rasterio numpy
"""
import rasterio
from rasterio.windows import Window
import numpy as np

def compute_sui_windowed(input_tif_path, output_tif_path, green_band=3, swir1_band=11):
    """
    Computes SUI = Green - SWIR1 using memory-efficient block windows.
    Handles NoData, zero-division, and writes DEFLATE-compressed output.
    """
    with rasterio.open(input_tif_path) as src:
        profile = src.profile.copy()
        nodata_val = src.nodata if src.nodata is not None else -9999.0
        
        # Configure output profile for single float32 index band
        profile.update(
            dtype=rasterio.float32,
            count=1,
            nodata=nodata_val,
            compress='deflate',
            tiled=True,
            blockxsize=512,
            blockysize=512
        )
        
        with rasterio.open(output_tif_path, 'w', **profile) as dst:
            # Iterate through native block windows to maintain small RAM footprint
            for _, window in src.block_windows(1):
                green = src.read(green_band, window=window).astype(np.float32)
                swir1 = src.read(swir1_band, window=window).astype(np.float32)
                
                # Create mask for valid pixels (non-nodata and positive values)
                valid_mask = (green != src.nodata) & (swir1 != src.nodata) & (green > 0)
                
                # Compute SUI (Asad et al., 2026)
                sui_chunk = np.full(green.shape, nodata_val, dtype=np.float32)
                with np.errstate(divide='ignore', invalid='ignore'):
                    # Scale factor if raw Sentinel-2 reflectance is integer 10000
                    scale = 0.0001 if green.max() > 100 else 1.0
                    sui_chunk[valid_mask] = (green[valid_mask] - swir1[valid_mask]) * scale
                
                # Write window chunk to disk
                dst.write(sui_chunk, 1, window=window)
                
    print(f"Successfully processed windowed SUI raster: {output_tif_path}")

# Example invocation
if __name__ == "__main__":
    compute_sui_windowed("S2_Bari_multiband.tif", "S2_Bari_SUI.tif")

stratified_rf_training.py

Stratified Random Sampling & Random Forest Training Pipeline

Extracts stratified pixel training samples across LULC classes (Sealed Urban, Bare Soil, Water, Vegetation) and trains an optimized Random Forest classifier with spatial validation and confusion metrics.

Python (geopandas + scikit-learn) 
"""
Stratified Remote Sensing Random Forest Classifier
Author: Ahmad Asad | Remote Sensing & GeoAI
Requirements: pip install geopandas rasterio scikit-learn pandas
"""
import geopandas as gpd
import rasterio
import numpy as np
import pandas as pd
from sklearn.ensemble import RandomForestClassifier
from sklearn.model_selection import StratifiedKFold
from sklearn.metrics import classification_report, confusion_matrix

def train_lulc_random_forest(raster_path, training_points_shp, label_col='class_id'):
    # 1. Load ground truth vector points
    gdf = gpd.read_file(training_points_shp)
    
    # 2. Sample multi-spectral raster values at point geometries
    coord_list = [(x, y) for x, y in zip(gdf.geometry.x, gdf.geometry.y)]
    with rasterio.open(raster_path) as src:
        sampled_values = [v for v in src.sample(coord_list)]
        band_names = [f"Band_{i+1}" for i in range(src.count)]
        
    X = pd.DataFrame(sampled_values, columns=band_names)
    y = gdf[label_col].values
    
    # 3. Stratified K-Fold Spatial Validation (k=5)
    skf = StratifiedKFold(n_splits=5, shuffle=True, random_state=42)
    rf = RandomForestClassifier(
        n_estimators=200,
        max_depth=18,
        min_samples_split=4,
        oob_score=True,
        n_jobs=-1,
        random_state=42
    )
    
    fold_accuracies = []
    for train_idx, val_idx in skf.split(X, y):
        X_train, X_val = X.iloc[train_idx], X.iloc[val_idx]
        y_train, y_val = y[train_idx], y[val_idx]
        
        rf.fit(X_train, y_train)
        score = rf.score(X_val, y_val)
        fold_accuracies.append(score)
        
    print(f"5-Fold Mean Accuracy: {np.mean(fold_accuracies):.4f} (+/- {np.std(fold_accuracies):.4f})")
    print(f"Out-of-Bag (OOB) Score: {rf.oob_score_:.4f}")
    
    # Feature Importance Analysis
    feat_importances = pd.Series(rf.feature_importances_, index=band_names).sort_values(ascending=False)
    print("\nFeature Importances:\n", feat_importances)
    
    return rf

openeo_datacube_composite.py

Cloud-Free Temporal Median Composite Datacube Export

Demonstrates standardized, vendor-neutral datacube processing via openEO. Connects to the Copernicus Data Space Ecosystem, masks clouds, reduces across the temporal dimension, and exports a GeoTIFF.

Python (openeo client) 
"""
openEO Datacube Temporal Median Pipeline
Author: Ahmad Asad | Remote Sensing & GeoAI
Backend: Copernicus Data Space Ecosystem (CDSE)
Requirements: pip install openeo
"""
import openeo

# 1. Connect and authenticate with openEO backend
conn = openeo.connect("https://openeo.dataspace.copernicus.eu").authenticate_oidc()

# 2. Define spatial & temporal bounds
bbox = {
    "west": 16.75, "east": 17.05,
    "south": 41.05, "north": 41.25,
    "crs": "EPSG:4326"
}
temporal_extent = ["2025-06-01", "2025-08-31"]

# 3. Load Sentinel-2 L2A optical and SCL bands
datacube = conn.load_collection(
    "SENTINEL_2_L2A",
    spatial_extent=bbox,
    temporal_extent=temporal_extent,
    bands=["B03", "B04", "B08", "B11", "SCL"]
)

# 4. Mask clouds and shadows using SCL band (classes 3, 8, 9, 10, 11)
scl = datacube.band("SCL")
cloud_mask = (scl == 3) | (scl == 8) | (scl == 9) | (scl == 10) | (scl == 11)
clean_cube = datacube.mask(cloud_mask)

# Drop SCL band and reduce temporal dimension with median reducer
optical_cube = clean_cube.filter_bands(["B03", "B04", "B08", "B11"])
median_mosaic = optical_cube.reduce_dimension(dimension="t", reducer="median")

# 5. Execute as batch job or download result
median_mosaic.download("Bari_S2_Summer_Median.tif", format="GTiff")
print("openEO processing completed.")

postgis_pv_cadastral_clean.sql

Idempotent Spatial Table Creation, GIST Indexing & Geometry Cleaning

Production SQL used in the Regione Puglia solar photovoltaic mapping project. Enforces CRS constraints, repairs self-intersecting cadastral polygons using ST_MakeValid, extracts valid multi-polygons, and creates GIST spatial indexes.

PostgreSQL / PostGIS SQL 
-- ============================================================================
-- Idempotent Spatial Schema, Geometry Repair & Spatial Indexing
-- Project: Regione Puglia Solar Photovoltaic Landscape Analysis
-- Author: Ahmad Asad (PhD Scholar · University of Bari Aldo Moro)
-- ============================================================================

-- 1. Ensure PostGIS extension is installed
CREATE EXTENSION IF NOT EXISTS postgis;

-- 2. Create target table with standardized EPSG:32633 (UTM zone 33N / Puglia)
CREATE TABLE IF NOT EXISTS solar_pv_parcels (
    parcel_id VARCHAR(64) PRIMARY KEY,
    municipality VARCHAR(128) NOT NULL,
    capacity_mw NUMERIC(8, 3),
    installation_year INT,
    orig_geom GEOMETRY(Geometry, 32633),
    clean_geom GEOMETRY(MultiPolygon, 32633),
    area_hectares NUMERIC(10, 4),
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- 3. Idempotently repair invalid, self-intersecting polygons from raw CAD/Shapefile
UPDATE solar_pv_parcels
SET clean_geom = ST_Multi(
    ST_CollectionExtract(
        ST_MakeValid(orig_geom),
        3 -- Extract Polygon components only (discards line/point slivers)
    )
)
WHERE clean_geom IS NULL 
  AND orig_geom IS NOT NULL;

-- 4. Calculate accurate territorial area in hectares
UPDATE solar_pv_parcels
SET area_hectares = ROUND((ST_Area(clean_geom) / 10000.0)::numeric, 4)
WHERE clean_geom IS NOT NULL;

-- 5. Build High-Performance GIST Spatial Index
CREATE INDEX IF NOT EXISTS idx_solar_pv_parcels_geom 
ON solar_pv_parcels USING GIST (clean_geom);

-- 6. Spatial Intersection: Aggregate agricultural land displaced by solar plants
SELECT 
    p.municipality,
    COUNT(p.parcel_id) AS total_plants,
    ROUND(SUM(p.area_hectares), 2) AS total_pv_hectares,
    ROUND(SUM(p.capacity_mw), 2) AS total_capacity_mw
FROM solar_pv_parcels p
WHERE ST_IsValid(p.clean_geom)
GROUP BY p.municipality
ORDER BY total_pv_hectares DESC
LIMIT 20;
